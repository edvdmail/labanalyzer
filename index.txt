function getFileType(url){
  const u=(url||'').toLowerCase().split('?')[0];
  if(u.endsWith('.pdf')) return 'pdf';
  if(u.match(/\.(png|jpg|jpeg|gif|webp|bmp|svg)$/)) return 'image';
  return 'other';
}
function getFileIcon(name){
  const n=(name||'').toLowerCase();
  if(n.includes('.pdf')) return '📄';
  if(n.match(/\.(png|jpg|jpeg|gif|webp)$/)) return '🖼️';
  return '📋';
}
function esc(s){return(s||'').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;')}
function decodeAltName(s){
  if(!s) return s;
  const parts=(s||'').split('_');
  const last=parts[parts.length-1];
  if(/^\d{8}$/.test(last)){
    const d=last.replace(/(\d{4})(\d{2})(\d{2})/,'$1/$2/$3');
    return parts.slice(0,-1).join(' ')+' · '+d;
  }
  return parts.join(' ');
}

═══════════════════════════════ */
function applyFilter(estado){
  activeFilter=estado;
  document.querySelectorAll('.chip.clickable').forEach(ch=>{
    const f=ch.dataset.filter;
    ch.classList.toggle('active-filter', f===estado);
    ch.classList.toggle('faded', f!==estado && estado!=='all' && f!=='all');
  });
  document.querySelectorAll('.pRow[data-estado]').forEach(row=>{
    row.classList.toggle('hidden', estado!=='all' && row.dataset.estado!==estado);
  });

  // Reconstruir chart con el examen activo filtrado
  const exam = allExamenes[examIdx];
  if(!exam) return;
  const res = exam.resultados || [];
  const filtered = estado==='all'
    ? res.filter(r=>!isNaN(parseFloat(r.valor)))
    : res.filter(r=>r.estado===estado && !isNaN(parseFloat(r.valor)));
  const chartBox=document.getElementById('chartBox');
  if(filtered.length===0){ chartBox.style.display='none'; if(chartInst){chartInst.destroy();chartInst=null;} return; }
  chartBox.style.display='block';
  if(chartInst) chartInst.destroy();
  chartInst=new Chart(document.getElementById('myChart').getContext('2d'),{
    type:'bar',
    data:{labels:filtered.map(r=>r.parametro),datasets:[{
      data:filtered.map(r=>parseFloat(r.valor)),
      backgroundColor:filtered.map(r=>(C[r.estado]||C.sin_dato).bg),
      borderColor:filtered.map(r=>(C[r.estado]||C.sin_dato).b),
      borderWidth:1.5,borderRadius:5,borderSkipped:false}]},
    options:{responsive:true,maintainAspectRatio:false,
      plugins:{legend:{display:false},tooltip:{
        backgroundColor:'#fff',titleColor:'#0d1929',bodyColor:'#3d5068',
        borderColor:'#dde4ed',borderWidth:1,padding:10,
        callbacks:{title:it=>it[0].label,label:it=>{
          const r=filtered[it.dataIndex];
          return[` ${t('tipV')}: ${r.valor}${r.unidad?' '+r.unidad:''}`,
                 ` ${t('tipR')}: ${r.referencia||'N/A'}`,
                 ` ${t('tipS')}: ${stateL(r.estado).toUpperCase()}`];}}}},
      scales:{
        x:{ticks:{color:'#8fa3bc',font:{family:"'DM Mono',monospace",size:9},maxRotation:40},grid:{color:'#f0f4f8'}},
        y:{ticks:{color:'#8fa3bc',font:{family:"'DM Mono',monospace",size:9}},grid:{color:'#f0f4f8'}}
      }}
  });
}

function zoom(dir){
  const steps=[0.5,0.75,1,1.25,1.5,2,2.5,3];
  const cur=steps.findIndex(s=>Math.abs(s-zoomLevel)<0.01);

  if(dir===0){ zoomLevel=1; }
  else if(dir>0){ zoomLevel=steps[Math.min(cur+1,steps.length-1)]; }
  else          { zoomLevel=steps[Math.max(cur-1,0)]; }

  document.getElementById('zoomLabel').textContent=Math.round(zoomLevel*100)+'%';

  if(currentFileType==='image'){

    if(img) img.style.transform=`scale(${zoomLevel})`;
  } else if(currentFileType==='pdf'){
    // Re-render proxy URL con zoom no aplica a iframe directamente,
    // pero actualizamos el label visualmente
    // Para PDF el zoom nativo del visor del browser maneja esto
  }
}

async function buscarArchivos(){
  const ident=document.getElementById('inputIdent').value.trim();
  if(!ident){alert(t('alertEmpty'));return;}
  const btn=document.getElementById('btnSearch');
  const lbl=document.getElementById('searchLabel');
  btn.disabled=true;btn.classList.add('loading');
  lbl.textContent=t('btnSearching');
  clearPreview();clearResults();selectedIdx=null;
  try{
    const res=await fetch('/archivos',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({identificacion:ident})});
    const data=await res.json();
    if(!res.ok) throw new Error(data.detail||'Error en consulta');
    currentFiles=data.archivos||[];
    renderFileList(currentFiles);
  }catch(e){currentFiles=[];renderFileList([],e.message);}
  finally{btn.disabled=false;btn.classList.remove('loading');lbl.textContent=t('btnSearch');}
}

/* ═══════════════════════════════
   RENDER FILE LIST
═══════════════════════════════ */
function renderFileList(files,errorMsg){
  const list=document.getElementById('fileList');
  list.innerHTML='';
  if(errorMsg){
    list.innerHTML=`<div class="emptyState"><div class="bigEm">⚠️</div><p style="color:var(--err)">${errorMsg}</p></div>`;
    return;
  }
  if(!files.length){
    list.innerHTML=`<div class="emptyState"><div class="bigEm">📭</div><p>${t('emptyNoFiles').replace(/\n/g,'<br>')}</p></div>`;
    return;
  }
  files.forEach((f,i)=>{
    const done = f.ya_analizado === true;
    const displayName = done && f.nombre_alternativo
      ? decodeAltName(f.nombre_alternativo)
      : f.nombre_archivo;
    const icon = getFileIcon(f.nombre_archivo||'');
    const el = document.createElement('div');
    el.className = 'fileItem' + (done ? ' done' : '');
    el.dataset.idx = i;
    const eyeAction = done ? `loadStoredAnalysis(${i})` : `previewFile(${i})`;
    const eyeTitle  = done ? t('alreadyAnalyzed') : 'Vista previa';
    el.innerHTML=`
      <span class="fileIcon">${icon}</span>
      <div class="fileMeta">
        <div class="fileName" title="${esc(displayName)}">${esc(displayName||'Archivo')}</div>
        ${!done ? `<div class="fileDate">${f.fecha_cargue||'—'}</div>` : ''}
      </div>
      <div class="fileActions">
        <button class="btnIconSm" title="${eyeTitle}"
          onclick="event.stopPropagation();${eyeAction}">👁</button>
        <button class="btnIconSm analyzeBtn${done?' analyzeBtn':''}" id="analyzeBtn-${i}"
          title="${done ? t('alreadyAnalyzed') : 'Analizar'}"
          ${done ? 'disabled' : ''}
          onclick="event.stopPropagation();analyzeFile(${i})">⚡</button>
      </div>
      <span class="doneBadge">✓ ${t('alreadyAnalyzed')}</span>
      <span class="analyzingBadge">${t('analyzing')}</span>`;
    el.addEventListener('click',()=>{
      if(done && f.analisis){ loadStoredAnalysis(i); }
      else { previewFile(i); }
    });
    list.appendChild(el);
  });
}

  currentFileType=ftype;
  currentFileUrl=f.url;
  zoomLevel=1;
  document.getElementById('zoomLabel').textContent='100%';

  const tp=document.getElementById('previewType');
  tp.textContent=ftype.toUpperCase();

  const body=document.getElementById('previewBody');
  body.innerHTML=`<div class="previewEmpty"><div class="pIcon">⏳</div><span>${t('loadingPreview')}</span></div>`;

  // Mostrar zoom solo para imágenes (PDF usa zoom nativo del browser)
  document.getElementById('zoomBar').style.display=(ftype==='image')?'flex':'none';

  if(ftype==='pdf'){
    loadPDF(f.url, body);
  } else if(ftype==='image'){

  } else {
    body.innerHTML=`<div class="previewEmpty"><div class="pIcon">📄</div><span>${t('noPreview')}</span></div>`;
  }
}

function loadPDF(url, body){
  // Intentar primero con el proxy propio (resuelve CORS y X-Frame-Options)
  const proxyUrl=`/proxy-file?url=${encodeURIComponent(url)}`;
  const iframe=document.createElement('iframe');
  iframe.title='PDF Preview';
  iframe.style.cssText='width:100%;height:100%;border:none;display:block';

  // Detectar si el iframe falla (timeout o error de carga)
  let loaded=false;
  const fallbackTimer=setTimeout(()=>{
    if(!loaded) showPDFFallback(url, body);
  }, 8000);

  iframe.onload=()=>{
    loaded=true;
    clearTimeout(fallbackTimer);
    // Verificar si el contenido es válido (algunos servers devuelven 200 con HTML de error)
    try{
      const doc=iframe.contentDocument||iframe.contentWindow?.document;
      if(doc && doc.body && doc.body.innerText && doc.body.innerText.length<100 && !doc.body.querySelector('embed')){
        showPDFFallback(url, body);
      }
    }catch(e){ /* cross-origin, probablemente OK */ }
  };
  iframe.onerror=()=>{ clearTimeout(fallbackTimer); showPDFFallback(url, body); };
  iframe.src=proxyUrl;
  body.innerHTML='';
  body.appendChild(iframe);
}

function showPDFFallback(url, body){
  const gdocsUrl=`https://docs.google.com/gview?url=${encodeURIComponent(url)}&embedded=true`;
  const proxyUrl=`/proxy-file?url=${encodeURIComponent(url)}`;

  body.innerHTML=`
    <div class="pdfFallback">
      <div class="fIco">📄</div>
      <p>${t('pdfBlocked')}</p>
      <div class="fallbackBtns">
        <button class="btnFallback primary" onclick="loadGDocsViewer('${esc(url)}')">
          📖 ${t('openGdocs')}
        </button>
        <a class="btnFallback" href="${esc(proxyUrl)}" target="_blank">
          🔗 ${t('openDirect')}
        </a>
        <a class="btnFallback" href="${esc(proxyUrl)}" download>
          ⬇ ${t('download')}
        </a>
      </div>
    </div>`;
}

function loadGDocsViewer(url){
  const gdocsUrl=`https://docs.google.com/gview?url=${encodeURIComponent(url)}&embedded=true`;
  const body=document.getElementById('previewBody');
  body.innerHTML=`<iframe src="${gdocsUrl}" style="width:100%;height:100%;border:none;display:block" title="PDF via Google Docs"></iframe>`;
  document.getElementById('zoomBar').style.display='none';
}

function loadImage(url, body){
  const wrap=document.createElement('div');
  wrap.className='imgZoomWrap';
  const img=new Image();
  img.onload=()=>{
    img.style.transformOrigin='top center';
    img.style.transition='transform .2s';
    wrap.appendChild(img);
    body.innerHTML='';
    body.appendChild(wrap);
    // Pinch to zoom (móvil)
    enablePinchZoom(wrap, img);
  };
  img.onerror=()=>{
    body.innerHTML=`<div class="previewEmpty"><div class="pIcon">🖼️</div><span>${t('noPreview')}</span></div>`;
  };
  img.src=url;
  img.alt='preview';
}

/* ── Pinch-to-zoom para móvil ── */
function enablePinchZoom(wrap, img){
  let lastDist=0;
  wrap.addEventListener('touchstart',e=>{
    if(e.touches.length===2) lastDist=Math.hypot(
      e.touches[0].clientX-e.touches[1].clientX,
      e.touches[0].clientY-e.touches[1].clientY);
  },{passive:true});
  wrap.addEventListener('touchmove',e=>{
    if(e.touches.length===2){
      const dist=Math.hypot(
        e.touches[0].clientX-e.touches[1].clientX,
        e.touches[0].clientY-e.touches[1].clientY);
      if(lastDist>0){
        zoomLevel=Math.min(3,Math.max(0.5,zoomLevel*(dist/lastDist)));
        img.style.transform=`scale(${zoomLevel})`;
        document.getElementById('zoomLabel').textContent=Math.round(zoomLevel*100)+'%';
      }
      lastDist=dist;
    }
  },{passive:true});
}

/* ═══════════════════════════════
   ANALYZE FILE
═══════════════════════════════ */
async function analyzeFile(idx){
  await runAnalyze(idx, null);
}

async function runAnalyze(idx, password){
  const f = currentFiles[idx];
  if(!f||!f.url) return;

  // ── Guarda anti-reprocesamiento en el frontend ──
  // Si el archivo ya está marcado como analizado y tiene datos en memoria,
  // simplemente cargamos el análisis almacenado sin llamar al backend.
  if(f.ya_analizado && f.analisis && !password){
    loadStoredAnalysis(idx);
    return;
  }

  if(selectedIdx!==idx) previewFile(idx);
  const item = document.querySelector(`.fileItem[data-idx="${idx}"]`);
  const abtn = document.getElementById(`analyzeBtn-${idx}`);
  if(item) item.classList.add('analyzing');
  if(abtn) abtn.classList.add('analyzing');
  clearResults();
  document.getElementById('errBox').classList.remove('open');
  try{
    const bodyObj = {url:f.url, id_usuario:f.id_usuario||0, id_archivo:f.id_archivo||null, output_lang:lang, force_reprocess:false};
    if(password) bodyObj.password = password;
    const res = await fetch('/analyze',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(bodyObj)});

    if(res.status === 423){
      if(item) item.classList.remove('analyzing');
      if(abtn) abtn.classList.remove('analyzing');
      openPasswordModal(idx);
      return;
    }

    const raw = await res.text();
    let data;
    try{ data=JSON.parse(raw); }catch{ throw new Error((lang==='es'?'Respuesta no válida:\n':'Invalid response:\n')+raw); }

    if(res.status===400 && data.detail && data.detail.includes('Contraseña incorrecta')){
      if(item) item.classList.remove('analyzing');
      if(abtn) abtn.classList.remove('analyzing');
      openPasswordModal(idx);
      const errEl = document.getElementById('pwdErr');
      errEl.textContent = t('pwdWrong');
      errEl.classList.add('show');
      return;
    }

    if(data.error) throw new Error(data.error+(data.raw_output?'\n\n'+data.raw_output:''));

    // Normalizar al nuevo formato si viene legacy (sin "examenes")
    if(!data.examenes){
      data = { examenes:[data], id_usuario:data.id_usuario, id_archivo:data.id_archivo,
               multi:false, nombre_alternativo:data.nombre_alternativo };
    }

    renderMultiResults(data);

    // Si vino del caché del backend, mostrar toast
    if(data.cached){
      showToast(lang==='es' ? '📦 Cargado desde caché (ya analizado)' : '📦 Loaded from cache (already analyzed)');
    }

    // ── Guardar automáticamente en Oracle solo si es análisis nuevo ──
    if(!data.cached && f.id_archivo && data.nombre_alternativo){
      await saveAnalysis(f.id_archivo, data.nombre_alternativo, data, idx);
    }

  }catch(err){
    document.getElementById('errMsg').textContent=err.message;
    document.getElementById('errBox').classList.add('open');
  }finally{
    if(item) item.classList.remove('analyzing');
    if(abtn) abtn.classList.remove('analyzing');
  }
}

/* ── Guardar análisis en Oracle y actualizar item en lista ── */
async function saveAnalysis(idArchivo, nombreAlternativo, data, idx){
  try{
    const res = await fetch('/save-analysis',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({
        id_archivo: idArchivo,
        nombre_alternativo: nombreAlternativo,
        analisis: JSON.stringify(data)
      })
    });
    if(!res.ok){ console.warn('save-analysis error:', await res.text()); return; }

    currentFiles[idx].ya_analizado = true;
    currentFiles[idx].nombre_alternativo = nombreAlternativo;
    currentFiles[idx].analisis = JSON.stringify(data);

    const item = document.querySelector(`.fileItem[data-idx="${idx}"]`);
    if(item){
      item.classList.add('done');
      const nameEl = item.querySelector('.fileName');
      if(nameEl){
        const parts = nombreAlternativo.split('_');
        const datePart = parts[parts.length-1];
        const isDate = /^\d{8}$/.test(datePart);
        const namePart = isDate ? parts.slice(0,-1).join(' ') : parts.join(' ');
        nameEl.textContent = isDate ? namePart + ' · ' + datePart.replace(/(\d{4})(\d{2})(\d{2})/,'$1/$2/$3') : nombreAlternativo;
        nameEl.title = nombreAlternativo;
      }
      const abtn = document.getElementById(`analyzeBtn-${idx}`);
      if(abtn){ abtn.disabled = true; abtn.classList.add('analyzeBtn'); }
      const doneBadge = item.querySelector('.doneBadge');
      if(doneBadge) doneBadge.style.display='inline-flex';
      item.onclick = ()=>loadStoredAnalysis(idx);
    }

    showToast(t('savedOk'));
  }catch(e){
    console.warn('Error guardando análisis:', e);
  }
}

/* ── Cargar análisis almacenado ── */
function loadStoredAnalysis(idx){
  const f = currentFiles[idx];
  if(!f||!f.analisis) return;
  previewFile(idx);
  try{
    let data = typeof f.analisis === 'string' ? JSON.parse(f.analisis) : f.analisis;
    // Normalizar legacy
    if(!data.examenes){
      data = { examenes:[data], id_usuario:data.id_usuario, id_archivo:data.id_archivo,
               multi:false, nombre_alternativo:data.nombre_alternativo };
    }
    // Retrocompatibilidad: reconstruir tipo_examen desde nombre_alternativo si falta
    data.examenes.forEach(e => {
      if(!e.tipo_examen && data.nombre_alternativo){
        const parts = data.nombre_alternativo.split('_');
        const last = parts[parts.length-1];
        const nameOnly = /^\d{8}$/.test(last) ? parts.slice(0,-1) : parts;
        e.tipo_examen = nameOnly.join(' ');
      }
    });
    renderMultiResults(data);
  }catch(e){
    document.getElementById('errMsg').textContent = 'Error cargando análisis guardado: ' + e.message;
    document.getElementById('errBox').classList.add('open');
  }
}

/* ═══════════════════════════════
   RENDER — ENTRY POINT MULTI
═══════════════════════════════ */
function renderMultiResults(data){
  lastData = data;
  allExamenes = data.examenes || [];
  examIdx = 0;

  const multi = allExamenes.length > 1;

  // Banner multi-examen
  const banner = document.getElementById('multiExamBanner');
  const navWrap = document.getElementById('examNavWrap');
  const navBar  = document.getElementById('examNavBar');

  if(multi){
    document.getElementById('multiExamTitle').textContent =
      lang==='es' ? `📋 ${allExamenes.length} exámenes detectados en este documento`
                  : `📋 ${allExamenes.length} exams detected in this document`;
    document.getElementById('multiExamSub').textContent =
      lang==='es' ? 'Navega entre ellos con los botones de abajo.'
                  : 'Navigate between them using the buttons below.';
    banner.style.display='flex';

    // Botones de navegación — nombre completo visible, barra con scroll
    navBar.innerHTML = allExamenes.map((e,i)=>{
      const nombre = e.tipo_examen || (lang==='es'?'Examen':'Exam');
      return `<button class="examNavBtn${i===0?' active':''}" id="examNavBtn-${i}"
        onclick="switchExam(${i})" title="${esc(nombre)}">
        <span class="eIdx">${i+1}</span>
        ${esc(nombre)}
      </button>`;
    }).join('')+
      `<div class="examNavArrows">
        <button class="examNavArrow" id="navPrev" onclick="switchExam(examIdx-1)" title="Anterior">‹</button>
        <span class="examCounter" id="examCounter">1/${allExamenes.length}</span>
        <button class="examNavArrow" id="navNext" onclick="switchExam(examIdx+1)" title="Siguiente">›</button>
      </div>`;
    navWrap.style.display='block';
  } else {
    banner.style.display='none';
    navWrap.style.display='none';
  }

  document.getElementById('resultsArea').classList.add('open');
  renderExam(allExamenes[0], data);
}

function switchExam(idx){
  if(idx < 0 || idx >= allExamenes.length) return;
  examIdx = idx;

  // Actualizar botones nav activos
  document.querySelectorAll('.examNavBtn').forEach((b,i)=>b.classList.toggle('active',i===idx));
  document.getElementById('examCounter').textContent=`${idx+1}/${allExamenes.length}`;
  document.getElementById('navPrev').disabled = idx===0;
  document.getElementById('navNext').disabled = idx===allExamenes.length-1;

  renderExam(allExamenes[idx], lastData);

  // Scroll suave al área de resultados
  document.getElementById('resultsArea').scrollIntoView({behavior:'smooth', block:'nearest'});
}

/* ═══════════════════════════════
   RENDER — UN EXAMEN
═══════════════════════════════ */
function renderExam(exam, rootData){
  const res   = exam.resultados    || [];
  const notas = exam.notas_clinicas || [];

  // ── JSON tab ──
  document.getElementById('jsonOut').innerHTML = hlJSON(
    allExamenes.length > 1 ? exam : rootData
  );

  // ── Tarjeta profesional ──
  // Prioridad: profesional del examen > profesional_documento
  const profDoc  = rootData.profesional_documento || {};
  const instDoc  = rootData.institucion_documento  || '';

  const nombre   = exam.profesional          || profDoc.nombre   || '';
  const cargo    = exam.cargo_profesional     || profDoc.cargo    || '';
  const registro = exam.registro_profesional  || profDoc.registro || '';
  const inst     = exam.institucion           || instDoc          || '';

  const profCard = document.getElementById('profCard');
  if(nombre || inst){
    profCard.style.display = 'flex';
    profCard.classList.toggle('missing', !nombre);

    const avatarEmoji = getAvatarEmoji(cargo);
    document.getElementById('profAvatar').textContent    = avatarEmoji;
    document.getElementById('profNombre').textContent    = nombre || (lang==='es'?'Profesional no identificado':'Unidentified professional');
    document.getElementById('profCargo').textContent     = cargo  || '';
    document.getElementById('profCargo').style.display   = cargo  ? 'block' : 'none';

    const regEl = document.getElementById('profRegistro');
    if(registro){
      regEl.textContent    = (lang==='es' ? 'Reg. Prof.: ' : 'License: ') + registro;
      regEl.style.display  = 'block';
    } else {
      regEl.style.display  = 'none';
    }

    const instRow = document.getElementById('profInstRow');
    if(inst){
      document.getElementById('profInst').textContent = inst;
      instRow.style.display = 'flex';
    } else {
      instRow.style.display = 'none';
    }
  } else {
    profCard.style.display = 'none';
  }

  // ── Título y fecha del examen ──
  const cachedBadge = lastData.cached
    ? `<span class="cachedBadge">📦 ${lang==='es'?'Caché':'Cached'}</span>` : '';
  const fhirCat   = exam.fhir_category || '';
  const fhirChip  = fhirCat
    ? `<span class="fhirChip ${fhirCat}">${fhirCat}</span>` : '';
  const subcat    = exam.subcategoria || exam.disciplina || '';
  const subcatChip = subcat
    ? `<span class="metaChip">🗂 ${esc(subcat)}</span>` : '';

  document.getElementById('examTitle').innerHTML =
    esc(exam.tipo_examen || (lang==='es'?'Examen de Laboratorio':'Laboratory Report'))
    + ' ' + fhirChip + cachedBadge;
  document.getElementById('examDate').textContent =
    exam.fecha
      ? `📅 ${t('dateL')}: ${exam.fecha}  ·  👤 ${t('userL')}: ${rootData.id_usuario}`
      : `👤 ${t('userL')}: ${rootData.id_usuario}`;

  // CUPS chip
  const cupsCodigo = exam.codigo_cups;
  const cupsNombre = exam.nombre_cups;
  const cupsFuente = exam.cups_fuente || 'sin_match';
  const cupsScore  = exam.cups_score  || 0;
  let cupsChipHtml = '';
  if(cupsCodigo){
    const tt = `${cupsNombre||''} (score: ${cupsScore})`;
    const ico = cupsFuente==='llm' ? '🤖' : '🏷';
    cupsChipHtml = `<span class="cupsChip ${cupsFuente}" title="${esc(tt)}">${ico} CUPS ${cupsCodigo}</span>`;
  } else {
    cupsChipHtml = `<span class="cupsChip sin_match" title="Sin código CUPS">🏷 CUPS —</span>`;
  }

  // examMetaRow: subcategoría + CUPS
  document.getElementById('examMetaRow').innerHTML = subcatChip + cupsChipHtml;

  // ── Summary chips ──
  const cnt={normal:0,alto:0,bajo:0,sin_dato:0};
  res.forEach(r=>{if(r.estado in cnt)cnt[r.estado]++;});
  document.getElementById('summaryBar').innerHTML=
    `<div class="chip total clickable active-filter" data-filter="all" onclick="applyFilter('all')">`+
    `<div class="dot" style="background:var(--muted)"></div>${TR[lang].cParams(res.length)}</div>`+
    (cnt.normal?`<div class="chip normal clickable" data-filter="normal" onclick="applyFilter('normal')"><div class="dot"></div>${TR[lang].cNormal(cnt.normal)}</div>`:'')+
    (cnt.alto  ?`<div class="chip alto clickable" data-filter="alto" onclick="applyFilter('alto')"><div class="dot"></div>${TR[lang].cHigh(cnt.alto)}</div>`  :'')+
    (cnt.bajo  ?`<div class="chip bajo clickable" data-filter="bajo" onclick="applyFilter('bajo')"><div class="dot"></div>${TR[lang].cLow(cnt.bajo)}</div>`   :'');

  // ── Chart ──
  const numericVals = res.filter(r=>!isNaN(parseFloat(r.valor)));
  const chartBox    = document.getElementById('chartBox');
  if(numericVals.length>0){
    chartBox.style.display='block';
    if(chartInst) chartInst.destroy();
    chartInst=new Chart(document.getElementById('myChart').getContext('2d'),{
      type:'bar',
      data:{labels:numericVals.map(r=>r.parametro),datasets:[{
        data:numericVals.map(r=>parseFloat(r.valor)),
        backgroundColor:numericVals.map(r=>(C[r.estado]||C.sin_dato).bg),
        borderColor:numericVals.map(r=>(C[r.estado]||C.sin_dato).b),
        borderWidth:1.5,borderRadius:5,borderSkipped:false}]},
      options:{responsive:true,maintainAspectRatio:false,
        plugins:{legend:{display:false},tooltip:{
          backgroundColor:'#fff',titleColor:'#0d1929',bodyColor:'#3d5068',
          borderColor:'#dde4ed',borderWidth:1,padding:10,
          callbacks:{title:it=>it[0].label,label:it=>{
            const r=numericVals[it.dataIndex];
            return[` ${t('tipV')}: ${r.valor}${r.unidad?' '+r.unidad:''}`,
                   ` ${t('tipR')}: ${r.referencia||'N/A'}`,
                   ` ${t('tipS')}: ${stateL(r.estado).toUpperCase()}`];}}}},
        scales:{
          x:{ticks:{color:'#8fa3bc',font:{family:"'DM Mono',monospace",size:9},maxRotation:40},grid:{color:'#f0f4f8'}},
          y:{ticks:{color:'#8fa3bc',font:{family:"'DM Mono',monospace",size:9}},grid:{color:'#f0f4f8'}}
        }}
    });
  } else {
    chartBox.style.display='none';
    if(chartInst){ chartInst.destroy(); chartInst=null; }
  }

  // ── Params grid ──
  const notaByParam={};
  notas.forEach(n=>{if(n.parametro) notaByParam[n.parametro.toLowerCase()]=n;});

  document.getElementById('paramsGrid').innerHTML = res.map(r=>{
    const inlineNota = notaByParam[r.parametro.toLowerCase()];
    const hasNota = !!inlineNota;
    return `<div class="pRow${hasNota?' has-nota':''}" data-estado="${r.estado}">
      <div class="pMain">
        <div class="pName">${esc(r.parametro)}</div>
        <div class="pVal">${esc(r.valor)}${r.unidad?' <span style="color:var(--muted);font-size:10px">'+esc(r.unidad)+'</span>':''}</div>
        ${r.referencia?`<span style="font-family:var(--fM);font-size:9px;color:var(--muted)">${esc(r.referencia)}</span>`:''}
        <span class="bdg ${r.estado}">${stateL(r.estado)}</span>
      </div>
      ${r.metodo?`<div class="pMetodo">⚗️ ${esc(r.metodo)}</div>`:''}
      ${hasNota?`<div class="pNotaInline">${NOTA_ICONS[inlineNota.tipo]||'•'} ${esc(inlineNota.descripcion)}</div>`:''}
    </div>`;
  }).join('');

  // ── Notas generales ──
  const notasSection = document.getElementById('notasSection');
  const notasGrid    = document.getElementById('notasGrid');
  const notasGenerales = notas.filter(n=>!n.parametro);
  if(notasGenerales.length>0){
    const order={critico:0,advertencia:1,info:2,positivo:3};
    const sorted=[...notasGenerales].sort((a,b)=>(order[a.tipo]??9)-(order[b.tipo]??9));
    let notasHtml = sorted.map(n=>`
      <div class="notaCard ${n.tipo||'info'}">
        <div class="notaIcon">${NOTA_ICONS[n.tipo]||'ℹ️'}</div>
        <div class="notaBody">
          <div class="notaTitulo">${esc(n.titulo||'')}</div>
          <div class="notaDesc">${esc(n.descripcion||'')}</div>
        </div>
      </div>`).join('');

    // Notas médico / paciente del examen (nuevos campos)
    if(exam.notas_medico){
      notasHtml += `<div class="notaCard advertencia" style="margin-top:6px">
        <div class="notaIcon">🩺</div>
        <div class="notaBody">
          <div class="notaTitulo">${lang==='es'?'Para el médico':'For the physician'}</div>
          <div class="notaDesc">${esc(exam.notas_medico)}</div>
        </div></div>`;
    }
    if(exam.notas_paciente){
      notasHtml += `<div class="notaCard info" style="margin-top:6px">
        <div class="notaIcon">👤</div>
        <div class="notaBody">
          <div class="notaTitulo">${lang==='es'?'Para el paciente':'For the patient'}</div>
          <div class="notaDesc">${esc(exam.notas_paciente)}</div>
        </div></div>`;
    }

    notasGrid.innerHTML = notasHtml;
    notasSection.style.display='block';
  } else if(exam.notas_medico || exam.notas_paciente){
    // No hay notas_clinicas generales pero sí notas médico/paciente
    let notasHtml = '';
    if(exam.notas_medico){
      notasHtml += `<div class="notaCard advertencia">
        <div class="notaIcon">🩺</div>
        <div class="notaBody">
          <div class="notaTitulo">${lang==='es'?'Para el médico':'For the physician'}</div>
          <div class="notaDesc">${esc(exam.notas_medico)}</div>
        </div></div>`;
    }
    if(exam.notas_paciente){
      notasHtml += `<div class="notaCard info">
        <div class="notaIcon">👤</div>
        <div class="notaBody">
          <div class="notaTitulo">${lang==='es'?'Para el paciente':'For the patient'}</div>
          <div class="notaDesc">${esc(exam.notas_paciente)}</div>
        </div></div>`;
    }
    notasGrid.innerHTML = notasHtml;
    notasSection.style.display = 'block';
  } else {
    notasSection.style.display='none';
  }

  activeFilter='all';
}

/* Emoji de avatar según especialidad */
function getAvatarEmoji(cargo){
  if(!cargo) return '👨‍⚕️';
  const c = cargo.toLowerCase();
  if(c.includes('radiol') || c.includes('rx') || c.includes('imagen')) return '🩻';
  if(c.includes('bacteriol') || c.includes('laborat'))                  return '🔬';
  if(c.includes('ecograf') || c.includes('sonograf') || c.includes('ultrasoni')) return '📡';
  if(c.includes('patol'))      return '🧫';
  if(c.includes('cardiol'))    return '❤️';
  if(c.includes('neurol'))     return '🧠';
  if(c.includes('ginecol') || c.includes('obstet')) return '🩺';
  if(c.includes('pediatr'))    return '👶';
  if(c.includes('ortoped') || c.includes('traumat')) return '🦴';
  return '👨‍⚕️';
}

function copyAs(format){
  if(!lastData) return;
  const exam  = allExamenes[examIdx] || {};
  const res   = exam.resultados    || [];
  const notas = exam.notas_clinicas || [];
  const profDoc  = lastData.profesional_documento || {};
  const instDoc  = lastData.institucion_documento  || '';
  const examName = exam.tipo_examen || (lang==='es'?'Examen de Laboratorio':'Laboratory Report');
  const fecha    = exam.fecha || '—';

  // Datos del profesional para el examen activo
  const profNombre   = exam.profesional          || profDoc.nombre   || '';
  const profCargo    = exam.cargo_profesional     || profDoc.cargo    || '';
  const profRegistro = exam.registro_profesional  || profDoc.registro || '';
  const profInst     = exam.institucion           || instDoc          || '';

  let text='';

  if(format==='plain'){
    const lines=[`${examName} | ${t('dateL')}: ${fecha} | ${t('userL')}: ${lastData.id_usuario}`];
    if(profNombre)   lines.push(`${lang==='es'?'Profesional':'Professional'}: ${profNombre}${profCargo?' ('+profCargo+')':''}`);
    if(profRegistro) lines.push(`${lang==='es'?'Reg. Prof.':'License'}: ${profRegistro}`);
    if(profInst)     lines.push(`${lang==='es'?'Institución':'Institution'}: ${profInst}`);
    lines.push('', ...res.map(r=>`• ${r.parametro}: ${r.valor}${r.unidad?' '+r.unidad:''} [${stateL(r.estado).toUpperCase()}]${r.referencia?' ('+t('tipR')+': '+r.referencia+')':''}`));
    if(notas.length) lines.push('',`── ${t('aiSummaryTitle')} ──`,...notas.map(n=>`${NOTA_ICONS[n.tipo]||'•'} ${n.titulo}: ${n.descripcion}`));
    text=lines.join('\n');
  } else if(format==='table'){
    text=['Parámetro\tValor\tUnidad\tReferencia\tEstado',...res.map(r=>[r.parametro,r.valor,r.unidad||'',r.referencia||'',stateL(r.estado)].join('\t'))].join('\n');
  } else if(format==='markdown'){
    const rows=res.map(r=>`| ${r.parametro} | ${r.valor} | ${r.unidad||''} | ${r.referencia||''} | ${stateL(r.estado)} |`);
    const notaMd=notas.length?[`\n### ${t('aiSummaryTitle')}`,...notas.map(n=>`- **${n.titulo}**: ${n.descripcion}`)]:[];
    const profMd=profNombre?[`> **${lang==='es'?'Profesional':'Professional'}:** ${profNombre}${profCargo?' · '+profCargo:''}${profRegistro?' · Reg: '+profRegistro:''}${profInst?' · '+profInst:''}`,'']:[];
    text=[`## ${examName}`,`**${t('dateL')}:** ${fecha}  **${t('userL')}:** ${lastData.id_usuario}`,...profMd,'',
      `| Parámetro | ${t('tipV')} | Unidad | ${t('tipR')} | ${t('tipS')} |`,'|---|---|---|---|---|',
      ...rows,...notaMd].join('\n');
  } else {
    text=JSON.stringify(allExamenes.length > 1 ? lastData : exam, null, 2);
  }

  navigator.clipboard.writeText(text).then(()=>{
    const idMap={plain:'cPlain',table:'cTable',markdown:'cMarkdown',json:'cJSON'};
    const btn=document.getElementById(idMap[format]);
    if(btn){btn.classList.add('done');setTimeout(()=>btn.classList.remove('done'),1800);}
    showToast(t('copied'));
  });
}


function copyRawJSON(){
  if(!lastData) return;
  navigator.clipboard.writeText(JSON.stringify(lastData,null,2)).then(()=>{
    const btn=document.getElementById('btnCpJSON');
    btn.textContent=t('copied');btn.classList.add('cp');
    setTimeout(()=>{btn.textContent=t('copyBtn');btn.classList.remove('cp');},2000);
  });
}

/* ═══════════════════════════════

═══════════════════════════════ */
function hlJSON(obj){
  return JSON.stringify(obj,null,2).replace(
    /("(\\u[a-zA-Z0-9]{4}|\\[^u]|[^\\"])*"(\s*:)?|\b(true|false|null)\b|-?\d+(?:\.\d*)?(?:[eE][+\-]?\d+)?)/g,
    m=>{
      if(/^"/.test(m)) return/:$/.test(m)?`<span class="jk">${m}</span>`:`<span class="js">${m}</span>`;
      if(/true|false/.test(m)) return`<span class="jb">${m}</span>`;
      if(/null/.test(m))       return`<span class="jz">${m}</span>`;
      return`<span class="jn">${m}</span>`;
    }
  );
}
